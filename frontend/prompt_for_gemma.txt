import 'package:cloud_firestore/cloud_firestore.dart';

class PollingState {
  final String status;
  final String message;
  final String? result; // 最終的ななぞかけJSON文字列
  final bool isError;

  PollingState({
    required this.status,
    required this.message,
    this.result,
    this.isError = false,
  });
}

class TaskStreamService {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;

  /// Firestoreのドキュメント変更をリアルタイムに監視し、状態のStreamを返す
  Stream<PollingState> watchTaskStatus(String taskId) {
    return _firestore.collection('tasks').doc(taskId).snapshots().map((
      snapshot,
    ) {
      if (!snapshot.exists || snapshot.data() == null) {
        return PollingState(status: 'pending', message: 'タスクの開始を待機中...');
      }

      final data = snapshot.data()!;
      final status = data['status'] as String? ?? 'pending';
      final message = data['message'] as String? ?? '処理中...';

      if (status == 'completed') {
        return PollingState(
          status: status,
          message: message,
          result: data['result'] as String?, // 最終結果
        );
      } else if (status == 'failed') {
        return PollingState(status: status, message: message, isError: true);
      }

      // 進行中のステータス
      return PollingState(status: status, message: message);
    });
  }
}
