   def _get_top_level_names(code_str: str) -> set[str]:
       import ast
       tree = ast.parse(code_str)
       return {node.name for node in tree.body if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))}
